bool StretchHelixFormingProcess::
resolveActiveZoneBoundaryFrames(
    ActiveZoneBoundaryFrames& boundaries) const
{
    boundaries =
        ActiveZoneBoundaryFrames{};

    boundaries.entry =
        activeFormingFrame;

    const double totalFormedLength =
        std::clamp(
            state.wrappedLength,
            0.0,
            input.pipeArcLength
        );

    const double activeLength =
        std::min(
            activeZoneLength,
            totalFormedLength
        );

    if (activeLength <= 1e-12)
    {
        boundaries.exit =
            boundaries.entry;

        boundaries.valid =
            true;

        return true;
    }

    const SpatialCurveIntegrationResult* formingReference =
        &referenceResult;

    if (
        mechanicsValid
        && loadedReferenceResult.valid
        && loadedReferenceResult.isComplete()
    )
    {
        formingReference =
            &loadedReferenceResult;
    }

    if (!formingReference->valid
        || !formingReference->isComplete()
        || formingReference->nodes.size() < 2)
    {
        return false;
    }

    boundaries.exit =
        boundaries.entry;

    boundaries.exit.P =
        activeFormingFrame.P
        + activeFormingFrame.T
        * activeLength;

    const PipeNode& formedReferenceNode =
        formingReference->nodes[1];

    boundaries.exit.T =
        formedReferenceNode.T;

    boundaries.exit.N =
        formedReferenceNode.N;

    boundaries.exit.B =
        formedReferenceNode.B;

    Vec3D supportAxisDirection =
        activeFormingFrame.T
            * kinematics.torsion
        + activeFormingFrame.B
            * kinematics.curvature;

    if (supportAxisDirection.lengthSquared() < 1e-12)
        return false;

    supportAxisDirection =
        supportAxisDirection.normalized();

    boundaries.exit.T =
        RigidTransformUtils::rotateAroundAxis(
            boundaries.exit.T,
            supportAxisDirection,
            state.supportRotationAngle
        ).normalized();

    boundaries.exit.N =

            boundaries.exit.N,
            supportAxisDirection,
            state.supportRotationAngle

        RigidTransformUtils::rotateAroundAxis(
            boundaries.exit.B,

MH1.16.10 — Hermite equation

For normalized position:

u ∈ [0,1]

use:

P(u) =
    h00 P0
  + h10 M0
  + h01 P1
  + h11 M1

where:

h00 =  2u³ - 3u² + 1
h10 =    u³ - 2u² + u
h01 = -2u³ + 3u²
h11 =    u³ -   u²

And:

P0 = ACTIVE entry
P1 = ACTIVE exit


M0 = T_entry * tangentScale
M1 = T_exit  * tangentScale

For the first test:

tangentScale =
    activeLength;

    MH1.16.11 — Add local Hermite helper

Inside StretchHelixFormingProcess.cpp, anonymous namespace:

Vec3D evaluateHermitePosition(
    const Vec3D& p0,
    const Vec3D& p1,
    const Vec3D& m0,
    const Vec3D& m1,
    double u)
{
    const double u2 =
        u * u;


    const double u3 =
        u2 * u;


    const double h00 =
        2.0 * u3
        - 3.0 * u2
        + 1.0;


    const double h10 =
        u3
        - 2.0 * u2
        + u;


    const double h01 =
        -2.0 * u3
        + 3.0 * u2;


    const double h11 =
        u3
        - u2;


    return
        p0 * h00
        + m0 * h10
        + p1 * h01
        + m1 * h11;
}

MH1.16.12 — Add tangent derivative helper

For correct PipeNode.T:

Vec3D evaluateHermiteTangent(
    const Vec3D& p0,
    const Vec3D& p1,
    const Vec3D& m0,
    const Vec3D& m1,
    double u)
{
    const double u2 =
        u * u;


    const double dh00 =
        6.0 * u2
        - 6.0 * u;


    const double dh10 =
        3.0 * u2
        - 4.0 * u
        + 1.0;


    const double dh01 =
        -6.0 * u2
        + 6.0 * u;


    const double dh11 =
        3.0 * u2
        - 2.0 * u;


    return
        p0 * dh00
        + m0 * dh10
        + p1 * dh01
        + m1 * dh11;
}

Normalize the result before assigning it to a node.
